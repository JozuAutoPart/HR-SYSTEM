import cv2 # เปิดกล้องเว็บแคม cap = cv2.VideoCapture(0) print("--- ระบบถ่ายภาพต้นแบบ ---") print("1. วางชิ้นงาน OK ที่สมบูรณ์ไว้ใต้กล้องและจัดแสงให้คงที่") print("2. กดปุ่ม 'c' บนคีย์บอร์ดเพื่อบันทึกภาพ") print("3. กดปุ่ม 'q' เพื่อปิดโปรแกรม") while True: ret, frame = cap.read() if not ret: print("ไม่สามารถเชื่อมต่อกล้องได้") break # แสดงภาพสดจากกล้อง cv2.imshow("Capture Template - Press 'c' to Save", frame) key = cv2.waitKey(1) & 0xFF if key == ord("c"): # แปลงเป็นภาพขาวดำหรือใช้ภาพสีก็ได้ (ในที่นี้ใช้ภาพสีหรือตัดเฉพาะส่วน) cv2.imwrite("template_ok.jpg", frame) print("บันทึกภาพสำเร็จ! ได้ไฟล์ 'template_ok.jpg' เรียบร้อยแล้ว") break elif key == ord("q"): break cap.release() cv2.destroyAllWindows()
